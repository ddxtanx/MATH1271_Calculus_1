  \question[5] Compute the derivative of
  \begin{equation}
    p(q) = \dfrac{q^3}{\sin(q)+6q^8}.
  \end{equation}
  You need not expand the denominator of the resulting function.
  \begin{solution}
    \begin{align}
      \dfrac{d}{dq} p(q) &= \dfrac{d}{dq} \dfrac{q^3}{\sin(q)+6q^8} \\
                         &= \dfrac{\dfrac{d}{dq} (q^3) \cdot (\sin(q) + 6q^8) - q^3 \cdot \dfrac{d}{dq}(\sin(q) + 6q^8)}{(\sin(q) + 6q^8)^2} \\
                         &= \dfrac{3q^2 \cdot (\sin(q) + 6q^8) - q^3 \cdot (\cos(q) + 48q^7)}{(\sin(q) + 6q^8)^2} \\
                         &= \dfrac{3q^2 \sin(q) + 18q^{10} - q^3 \cos(q) + 48q^{10}}{(\sin(q)+6q^8)^2}
                         &= \dfrac{3q^2 \sin(q) - q^3 \cos(q) + 64 q^{10}}{(\sin(q) + 6q^8)^2}
    \end{align}
  \end{solution}
