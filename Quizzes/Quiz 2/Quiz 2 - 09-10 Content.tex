    \question Use the figure below, which gives a graph of the function $f(x)$, to give values for the indicated limits. If a limit does not exist, enter DNE.
    \begin{center}
      \includegraphics[width=0.3\textwidth]{QuizGraph.png}
      \nopagebreak
    \end{center}
    \begin{parts}
      \part[1] $\lim\limits_{x \rightarrow -2} f(x) = $
      \vspace{.2in}
      \part[3] $\lim\limits_{x \rightarrow 0} f(x) = $
      \vspace{.2in}

      \part[3] $\lim\limits_{x \rightarrow 2} f(x) = $
      \vspace{.2in}

      \part[3] $\lim\limits_{x \rightarrow 3} f(x) = $
      \vspace{.2in}
    \end{parts}
    \begin{solution}
      \begin{enumerate}
        \item $\lim\limits_{x \rightarrow -2} f(x) = 2$
        \item $\lim\limits_{x \rightarrow 0} f(x) = 6$
        \item $\lim\limits_{x \rightarrow 2} f(x)$ DNE since jump discontinuity
        \item $\lim\limits_{x \rightarrow 3} f(x) = 2$
      \end{enumerate}
    \end{solution}
