  \question[5] Differentiate the function:
  \begin{equation}
    h(t) = \sin(t^3 e^{-6t}).
  \end{equation}
  You need not distribute any products across sums, or otherwise simplify any quantities computed after resolving all differentiation.

  \begin{solution}
    \begin{align}
      h'(t) &= \dfrac{d}{dt}(\sin(t^3 e^{-6t})) \\
            &= (\dfrac{d}{dt}(t^3 e^{-6t})) \cdot \cos(t^3 e^{-6t}) \\
            &=(3t^2 e^{-6t} + t^3 \cdot \dfrac{d}{dt}(e^{-6t}))\cdot \cos(t^3 e^{-6t}) \\
            &=(3t^2 e^{-6t} + t^3 \cdot (\dfrac{d}{dt} -6t) \cdot e^{-6t}) \cdot \cos(t^3 e^{-6t}) \\
            &=(3t^2 e^{-6t} -6 t^3 e^{-6t}) \cos(t^3 e^{-6t})
    \end{align}
  \end{solution}
\end{questions}
